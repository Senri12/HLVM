void printInt(int x) {
    if (x < 0) {
        out('-');
        x = 0 - x;
    }

    if (x >= 10) {
        int prefix = x / 10;
        printInt(prefix);
    }

    int q = x / 10;
    q = q * 10;
    int digit = x - q;
    out(digit + '0');
}

void print(int arg) {
    printInt(arg);
    out('\n');
    return;
}

int readInt() {
    int ch = 0;
    int sign = 1;
    int x = 0;
    int tmp = 0;

    ch = readToken();

    if (ch == 0) {
        return 0;
    }

    if (ch == '-') {
        sign = 0 - 1;
        ch = readToken();
    }

    while (1) {
        if (ch < '0') {
            break;
        }

        if (ch > '9') {
            break;
        }

        x = x * 10;
        tmp = ch - '0';
        x = x + tmp;

        ch = in();
    }

    tmp = x * sign;
    return tmp;
}


int readToken() {
    while (1) {
        int ch = in();

        if (ch == 0) {
            return 0;
        }

        if (ch == 10) {
            continue;
        }

        if (ch == 13) {
            continue;
        }

        if (ch == 32) {
            continue;
        }

        if (ch == 9) {
            continue;
        }

        return ch;
    }
}

int main() {
    while (1) {
        out('>');

        int op = readToken();
        if (op == 0) {
            return 0;
        }

        int a = readInt();
        int b = readInt();
        if (a == 0) {
            return 0;
        }
        if (b == 0) {
            return 0;
        }

        if (op == '+') {
            int tmp = a + b;
            print(tmp);
            continue;
        }

        if (op == '-') {
            int tmp = a - b;
            print(tmp);
            continue;
        }

        if (op == '*') {
            int tmp = a * b;
            print(tmp);
            continue;
        }

        if (op == '/') {
            if (b == 0) {
                out('E');
                out('r');
                out('r');
                out('\n');
                continue;
            }

            int tmp = a / b;
            print(tmp);
            continue;
        }

        out('?');
        out('\n');
    }
}
